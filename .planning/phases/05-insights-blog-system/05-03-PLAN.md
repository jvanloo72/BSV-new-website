         ```
         <script is:inline>
           (function () {
             function readFilter() {
               const params = new URLSearchParams(window.location.search);
               return { author: params.get('author'), practice: params.get('practice') };
             }
             function applyFilter() {
               const { author, practice } = readFilter();
               const items = document.querySelectorAll('#post-list > li');
               let visible = 0;
               items.forEach(function (li) {
                 var a = li.getAttribute('data-author');
                 var p = li.getAttribute('data-practice');
                 var match = (!author || a === author) && (!practice || p === practice);
                 if (match) { li.removeAttribute('hidden'); visible++; }
                 else { li.setAttribute('hidden', ''); }
               });
               var emptyEl = document.getElementById('empty-filtered');
               var listEl = document.getElementById('post-list');
               if (emptyEl && listEl) {
                 if (visible === 0) { emptyEl.removeAttribute('hidden'); listEl.setAttribute('hidden',''); }
                 else { emptyEl.setAttribute('hidden',''); listEl.removeAttribute('hidden'); }
               }
               document.querySelectorAll('[data-chip]').forEach(function (chip) {
                 var param = chip.getAttribute('data-param');
                 var value = chip.getAttribute('data-value');
                 var current = param === 'author' ? author : practice;
                 var active = (value === null && !current) || value === current;
                 chip.setAttribute('aria-pressed', String(active));
               });
             }
             function onChipClick(e) {
               var chip = e.currentTarget;
               var href = chip.getAttribute('href');
               if (!href) return;
               e.preventDefault();
               history.pushState({}, '', href);
               applyFilter();
             }
             document.addEventListener('DOMContentLoaded', function () {
               applyFilter();
               document.querySelectorAll('[data-chip]').forEach(function (c) { c.addEventListener('click', onChipClick); });
               window.addEventListener('popstate', applyFilter);
             });
           })();
         </script>
         ```
         Use only `var`, `function`, and ES5-safe constructs inside the IIFE (the script runs without bundler transforms because of `is:inline`). NO template literals, NO arrow functions for shorter syntax (the `function () {}` form is explicit + transparent).
         The script never writes user input into `innerHTML`. It only sets attributes (`aria-pressed`, `hidden`) and reads from `window.location.search` — no XSS surface (T-05-02 mitigation).
